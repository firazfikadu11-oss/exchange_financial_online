-- Exchange Financial Online: supporting tables
-- Run in Supabase SQL Editor. Review RLS policies before public deployment.

create table if not exists public.withdrawal_requests (
  id bigint generated by default as identity primary key,
  user_id text,
  phone text not null,
  amount numeric(14,2) not null check (amount > 0),
  method text not null,
  account_details text not null,
  status text not null default 'pending' check (status in ('pending','approved','rejected','paid')),
  created_at timestamptz not null default now()
);

create index if not exists withdrawal_requests_phone_created_idx
  on public.withdrawal_requests (phone, created_at desc);

create table if not exists public.transactions (
  id bigint generated by default as identity primary key,
  phone text not null,
  type text not null,
  amount numeric(14,2) not null check (amount >= 0),
  status text not null default 'completed',
  created_at timestamptz not null default now()
);

create index if not exists transactions_phone_created_idx
  on public.transactions (phone, created_at desc);
